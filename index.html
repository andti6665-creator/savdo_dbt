Asosiy SQL VIEW va Protsedura kodlari

1. v_kunlik_savdo ko'rinishi:

CREATE OR REPLACE VIEW `v_kunlik_savdo` AS
SELECT
    `sana` AS `sana`,
    COUNT(DISTINCT `buyurtma_id`) AS `buyurtmalar`,
    SUM(`miqdor`) AS `jami_miqdor`,
    SUM(`summa`) AS `tushum`
FROM `v_buyurtma_tafsilot`
WHERE `holat` IN ('tolangan', 'yetkazildi')
GROUP BY `sana`;

2. sp_v_mk_insert saqlanuvchi protsedurasi:

DELIMITER $$
CREATE PROCEDURE `sp_v_mk_insert`(
    IN `p_nomi` VARCHAR(100),
    IN `p_kategoriya` VARCHAR(100),
    IN `p_narx` DECIMAL(12,2)
)
BEGIN
    INSERT IGNORE INTO kategoriyalar (nomi) VALUES (p_kategoriya);
    INSERT INTO mahsulotlar (nomi, kategoriya_id, narx)
    VALUES (
        p_nomi,
        (SELECT kategoriya_id FROM kategoriyalar WHERE nomi = p_kategoriya LIMIT 1),
        p_narx
    );
END$$
DELIMITER ;
